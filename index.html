function doPost(e) { try { var data = JSON.parse(e.postData.contents); // Se espera un array de arrays: [ [col1, col2...], [...] ] var ss = SpreadsheetApp.getActiveSpreadsheet(); // Procesar cada fila enviada data.forEach(function(row, index) { if (index === 0) return; // Omitir encabezado si viene incluido en las filas var area = row[3]; // Columna D (Índice 3: DEPENDENCIA ORIGEN EMISION) if (!area) area = "SIN_AREA"; // Sanitizar el nombre de la pestaña (máximo 31 caracteres) var sheetName = String(area).trim().substring(0, 30); var sheet = ss.getSheetByName(sheetName); if (!sheet) { sheet = ss.insertSheet(sheetName); // Insertar encabezados en la nueva pestaña sheet.appendRow(data[0]); } // Agregar la fila con todos los datos sheet.appendRow(row); }); return ContentService.createTextOutput(JSON.stringify({ status: "success", message: "Procesado correctamente" })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() })) .setMimeType(ContentService.MimeType.JSON); } }
